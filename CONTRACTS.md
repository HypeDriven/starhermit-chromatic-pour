# Chromatic Pour — Module Contracts

This file is the binding integration contract between modules. Every module is a
plain ES module (`"type":"module"`), usable in the browser; `rules.js`,
`rng.js`, `content.js`, `storage.js` and `session.js` must also run under
Node.js (no DOM, no `window`, no `localStorage` access at import time —
`storage.js` may touch `localStorage` only inside functions, guarded).

Three.js is vendored at `vendor/three.module.js` and mapped via import map as
`three`. Only `render.js` imports it.

## Color encoding

Colors are integer indices `0..colorCount-1`. Vessels are arrays of color
indices, **bottom first** (index 0 = bottom, last element = top).

## js/rules.js  (pure, deterministic, no I/O, no Date, no Math.random)

```js
export const RULES_VERSION = 1;
export const INVALID = {
  GAME_OVER:'game-over', OUT_OF_RANGE:'out-of-range', SAME_VESSEL:'same-vessel',
  SOURCE_EMPTY:'source-empty', DEST_FULL:'dest-full', COLOR_MISMATCH:'color-mismatch'
};
export const TERMINAL = {
  ALL_UNIFORM:'all-uniform', MOVE_LIMIT:'move-limit-exceeded',
  TIME_LIMIT:'time-limit-exceeded', ABANDONED:'abandoned'
};

createGame({seed, colorCount, capacity=4, emptyVessels=2, constraints={}}) -> state
// constraints: { moveLimit?:int, timeLimitMs?:int }
// Generated by scrambling a solved state with rulesStream(seed); guaranteed
// solvable and not already solved. capacity is always 4 for shipped content.
//
// state = {
//   version, seed, colorCount, capacity,
//   vessels: [[int,...], ...],
//   turn,             // monotonic, +1 per applied non-duplicate command
//   moves,            // completed pours
//   invalidActions,   // count of rejected pour attempts (recorded via recordInvalid)
//   elapsedMs,        // authoritative, supplied by session via cmd.elapsedMs
//   status: 'active'|'complete'|'failed',
//   terminalReason: null|TERMINAL.*,
//   constraints, appliedCommandIds: [string,...]
// }

topRun(state, i) -> {color, layers} | null        // top contiguous run of vessel i
canPour(state, from, to) -> {ok:true, color, layers} | {ok:false, reason:INVALID.*}
legalPours(state, {includePointless=false}={}) -> [{from,to,color,layers}]
// "Pointless" = pouring a uniform vessel into an empty vessel (pure shuffle).
applyCommand(state, cmd) -> {state, events, duplicate?} | {error:{reason, message}}
// cmd = {id:string, type:'pour', from, to, elapsedMs?:int}
// events = [{type:'pour',from,to,color,layers}] plus optional
//          {type:'complete'} or {type:'failed', reason}
// Idempotent: a repeated cmd.id returns the ORIGINAL result state with
// duplicate:true and no extra events. Invalid pours return {error} and do NOT
// mutate; the session then calls recordInvalid(state).
recordInvalid(state) -> state                      // invalidActions+1, turn unchanged
abandon(state) -> state                            // status 'failed', reason ABANDONED
isSolved(state) -> bool                            // every vessel empty or full-uniform
isVesselComplete(state, i) -> bool                 // uniform (any height) or empty
hint(state) -> {from,to,color,layers} | null       // solver-backed; same legal API
scoreState(state, parMoves) -> {
  total:int, components:{completion:int, moveEfficiency:int, timeBonus:int, invalidPenalty:int},
  breakdown:[{label:string, value:int}]            // signed values, total = sum
}
compareResults(a, b) -> int
// a/b = {complete:bool, invalidActions:int, elapsedMs:int, sessionId:string}
// ordering: completion, fewer invalid, lower elapsed, then sessionId. <0 = a wins.
serialize(state) -> string ; deserialize(str) -> state   // versioned, migrates
hashState(state) -> string                       // fnv1a hex over canonical fields
solve(startState, {maxNodes=200000}={}) -> {solvable:bool, depth:int|null, moves:[[from,to]]|null}
// BFS over pour graph, depth = minimum moves to solve (null if unsolvable/overflow).
```

## js/session.js  (Node-safe; clock injected)

```js
export class GameSession {
  constructor(levelDef, {sessionId, now} = {})
  // now: () => ms epoch. sessionId: stable string for tie-breaks (random per boot).
  start()                      // -> state; begins clock, status active
  pause() / resume()           // accumulates authoritative elapsedMs only while active
  elapsedMs() -> int
  selectVessel(i) -> {kind:'selected'|'pour', ...} | {kind:'error', reason, message}
  // selection state machine: first legal select lifts; second select pours or
  // re-selects; pouring from completed-uniform vessel is legal.
  pour(from, to) -> {ok:true, events} | {ok:false, reason, message}
  undo() -> bool               // restores prior snapshot; not counted as move;
                               // false if stack empty or level.constraints.noUndo
  restart()                    // back to initial state, same seed; history cleared
  canUndo -> bool ; selected -> int|null
  state -> immutable snapshot  // current rules state
  hint() -> {...}|null
  result() -> {score, complete, elapsedMs, moves, invalidActions, sessionId,
               hintsUsed, undosUsed, assists:{hints,undos}}
  checkTimeout() -> bool       // fails the round (time-limit-exceeded) when a
                               // timeLimitMs constraint has elapsed between pours
  replayEnvelope() -> {schemaVersion:1, build, contentVersion, seed, initialHash,
    startedAt, commands:[{id,type,from,to,elapsedMs}], stateHashes:[{turn,hash}],
    terminal:{status,reason,score}}
  snapshot() -> string ; static restore(json, {now}) -> GameSession  // durable reconnect
}
```

## js/ui.js + js/main.js (browser)

`main.js`: boot sequence, capability detection, import-map bootstrap, rAF loop
driving `renderer.renderFrame`, visibility/resize/orientation handling, global
error surface, app state machine:
`boot → title → profile-ready → mode-select → preparing → tutorial/countdown →
active ↔ paused → resolving → results → progression`. Every transition records
`{from,to,owner,reason}` in a debug log.

`ui.js` exports `createUI({root, session-agnostic services...})` and owns all
DOM: screens (title, mode select, setup, HUD, pause, results, help, settings,
progression, daily, score chase), the DOM vessel mirror (fully playable without
WebGL and the accessibility tree for the board), keyboard + gamepad input,
focus management, aria-live announcements. UI state is fully separate from
simulation state.

Input mapping (defaults, remappable list stored in settings.bindings):
Arrows/WASD move vessel focus; Enter/Space select+confirm; Esc pause/cancel;
U undo; H hint; R restart; C camera reset. Gamepad: dpad/left stick focus,
A confirm, B cancel, Start pause, X undo, Y hint.

## server.js  (Node, no dependencies; also the static dev server)

- Serves the distribution; `GET /api/v1/time` → `{now, epochMs}`.
- `POST /api/v1/scores` validates a replay envelope by re-simulating commands
  through `rules.js` (same seed/content version), rejects impossible or
  stale-version scores with `{"error":"..."}`; 429 on rate limit. The entry
  shape is the one listed under `platform.submitScore`; `board` must equal
  `entry.levelId`. Verifiable boards: today's daily (`daily-YYYY-MM-DD`,
  date-gated), authored challenges (fixed seeds/constraints), and score-chase
  boards (`practice-master-chase-YYYY-MM-DD|evergreen`).
- `GET /api/v1/leaderboard?board=<levelId>&scope=global|friends`.
- `POST /api/v1/achievements` — idempotent unlock delivery.
- JSON-file persistence under `data/` (created at runtime, git-ignored).
- Port from `PORT` env, default 8080. Run: `node server.js`.

## tests/run-tests.js

Node test runner (plain, no deps) covering: every legal action + invalid reason,
scoring components + tie ordering, serialization round-trip + migration,
replay determinism (same seed+commands ⇒ identical hashes), fuzz malformed
commands (no throws/hangs/NaN), content validation (all levels solvable, pars),
golden sessions (easy/medium/hard/interrupted/resumed/terminal).
