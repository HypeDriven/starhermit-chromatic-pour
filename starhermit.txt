# StarHermit engine script game: one package, played natively by the StarHermit
# client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine on
# wiki.starhermit.com by hash; never edit it). Re-run `npm run package` after
# every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.24.0
engine.minor=17
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# The shelf is also played with the pointer or touch: tap a vessel to lift
# it, tap another to pour.
control.focus_next=ArrowRight+KeyD | Focus next vessel
control.focus_prev=ArrowLeft+KeyA | Focus previous vessel
control.focus_up=ArrowUp+KeyW | Focus vessel above
control.focus_down=ArrowDown+KeyS | Focus vessel below
control.confirm=Enter+Space+NumpadEnter | Select / pour
control.cancel=Escape | Cancel selection / pause
control.pause=KeyP | Pause
control.undo=KeyU | Undo
control.hint=KeyH | Hint
control.restart=KeyR | Restart round
control.camera_reset=KeyC | Recenter view
engine.group.focus_next=Shelf
engine.group.pause=Round

# Gamepad defaults: the D-pad or left stick moves the focus over the vessels,
# A lifts and pours, B cancels, Start pauses, X undoes, Y hints.
engine.gamepad.focus_next=DpadRight+LeftStickRight
engine.gamepad.focus_prev=DpadLeft+LeftStickLeft
engine.gamepad.focus_up=DpadUp+LeftStickUp
engine.gamepad.focus_down=DpadDown+LeftStickDown
engine.gamepad.confirm=South
engine.gamepad.cancel=East
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.hint=North
engine.gamepad.restart=Select
engine.gamepad.camera_reset=RightStickPress
